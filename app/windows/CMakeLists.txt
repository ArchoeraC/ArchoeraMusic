# === ArchoeraMusic platform bridge：内嵌 CMake 构建链（一键构建，MSVC C++）===
# 产出 archoera_platform.dll 到 app/native/platform/build/out/；WinRT(SMTC/Toast)
# 需 cppwinrt 头：CMake 变量 CPPWINRT_INCLUDE 或 vcpkg cppwinrt CONFIG 包；
# 均缺失时 backend_windows.cpp 自动降级（SMTC 禁用、Toast 走 MessageBox）。
set(PLATFORM_BRIDGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../native/platform")
set(PLATFORM_BRIDGE_BUILD_DIR "${PLATFORM_BRIDGE_DIR}/build")
set(PLATFORM_BRIDGE_CMAKE_ARGS
  -S "${PLATFORM_BRIDGE_DIR}" -B "${PLATFORM_BRIDGE_BUILD_DIR}" -DCMAKE_BUILD_TYPE=Release)
# C++/WinRT 头路径：CMake 变量 CPPWINRT_INCLUDE 优先，其次环境变量（CI 以 env 传入）。
set(_cppwinrt_inc "${CPPWINRT_INCLUDE}")
if(NOT _cppwinrt_inc AND DEFINED ENV{CPPWINRT_INCLUDE})
  set(_cppwinrt_inc "$ENV{CPPWINRT_INCLUDE}")
endif()
if(_cppwinrt_inc)
  list(APPEND PLATFORM_BRIDGE_CMAKE_ARGS -DCPPWINRT_INCLUDE=${_cppwinrt_inc})
endif()
add_custom_target(platform_bridge_cmake ALL
  COMMAND ${CMAKE_COMMAND} ${PLATFORM_BRIDGE_CMAKE_ARGS}
  COMMAND ${CMAKE_COMMAND} --build "${PLATFORM_BRIDGE_BUILD_DIR}" --config Release -j
  COMMENT "cmake build archoera_platform bridge (MSVC C++)"
  VERBATIM
)
add_dependencies(${BINARY_NAME} platform_bridge_cmake)

install(FILES "${PLATFORM_BRIDGE_BUILD_DIR}/out/archoera_platform.dll"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime)

# === 原生库完整性清单（SEC-08）===
# 纯 CMake 生成（不依赖 bash/sh）：对 native/ 下全部文件，以及 exe 根
# archoera_music.exe / archoera_kernel.dll、data/app.so 计算 SHA-256，
# 写入 native/ARCHOERA-NATIVE-MANIFEST.txt，供 Dart 在 DynamicLibrary.open
# 前按 basename 校验（打包态 fail-closed，见 lib/services/native_integrity.dart）。
# 该 install(CODE) 声明于全部 native/ install 之后，故执行时文件已就位。
install(CODE "
  set(_nativedir \"\${CMAKE_INSTALL_PREFIX}/native\")
  file(MAKE_DIRECTORY \"\${_nativedir}\")
  set(_manifest \"\${_nativedir}/ARCHOERA-NATIVE-MANIFEST.txt\")
  set(_lines \"\")
  file(GLOB _files \"\${_nativedir}/*\")
  foreach(_f IN LISTS _files)
    if(IS_DIRECTORY \"\${_f}\")
      continue()
    endif()
    get_filename_component(_base \"\${_f}\" NAME)
    if(_base STREQUAL \"ARCHOERA-NATIVE-MANIFEST.txt\")
      continue()
    endif()
    file(SHA256 \"\${_f}\" _hash)
    list(APPEND _lines \"\${_hash}  \${_base}\")
  endforeach()
  foreach(_f \"\${CMAKE_INSTALL_PREFIX}/archoera_music.exe\"
            \"\${CMAKE_INSTALL_PREFIX}/archoera_kernel.dll\"
            \"\${CMAKE_INSTALL_PREFIX}/data/app.so\")
    if(EXISTS \"\${_f}\")
      get_filename_component(_base \"\${_f}\" NAME)
      file(SHA256 \"\${_f}\" _hash)
      list(APPEND _lines \"\${_hash}  \${_base}\")
    endif()
  endforeach()
  if(_lines)
    list(SORT _lines)
    string(ASCII 10 _nl)
    string(REPLACE \";\" \"\${_nl}\" _out \"\${_lines}\")
    file(WRITE \"\${_manifest}\" \"\${_out}\${_nl}\")
  endif()
" COMPONENT Runtime)

# === vcpkg 运行时 DLL（FFmpeg 等动态依赖）===
# audio-engine（archoera_mediaengine.dll / scraper）经 vcpkg import libs 动态链接
# ffmpeg/taglib/curl 等（triplet x64-windows-release 为动态构建）；运行时必须把
# vcpkg_installed/<triplet>/bin/*.dll 一并分发到 exe 根目录——Windows 加载器按
# exe 目录解析 archoera_mediaengine.dll 的依赖，缺 FFmpeg DLL 会直接加载失败
#（表现为播放稳定跳过所有歌曲）。triplet 与 build_windows.bat 对齐。
if(NOT DEFINED VCPKG_TARGET_TRIPLET)
  set(VCPKG_TARGET_TRIPLET "$ENV{VCPKG_TARGET_TRIPLET}")
  if(NOT VCPKG_TARGET_TRIPLET)
    set(VCPKG_TARGET_TRIPLET "x64-windows-release")
  endif()
endif()
set(VCPKG_BIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vcpkg_installed/${VCPKG_TARGET_TRIPLET}/bin")
file(GLOB VCPKG_RUNTIME_DLLS "${VCPKG_BIN_DIR}/*.dll")
if(VCPKG_RUNTIME_DLLS)
  install(FILES ${VCPKG_RUNTIME_DLLS}
    DESTINATION "${CMAKE_INSTALL_PREFIX}"
    COMPONENT Runtime)
endif()

# LGPL 合规：随包附 FFmpeg / TagLib 许可文本（vcpkg 的 copyright 文件）。
# vcpkg 的 ffmpeg 不开 openssl 特性 → Windows 自动走系统 **Schannel** 提供 https/tls，
# 且保持纯 LGPL-2.1+（不带 version3），与 AGPL-3.0 聚合分发兼容
# （见 app/core/audio-engine/THIRD-PARTY-LICENSES.md）。
set(VCPKG_SHARE_DIR
  "${CMAKE_CURRENT_SOURCE_DIR}/../vcpkg_installed/${VCPKG_TARGET_TRIPLET}/share")
foreach(_dep IN ITEMS ffmpeg taglib)
  if(EXISTS "${VCPKG_SHARE_DIR}/${_dep}/copyright")
    install(FILES "${VCPKG_SHARE_DIR}/${_dep}/copyright"
      DESTINATION "${CMAKE_INSTALL_PREFIX}/licenses"
      RENAME "${_dep}-LICENSE.txt"
      COMPONENT Runtime)
  endif()
endforeach()

# === mingw 运行时 DLL（subsonic Go c-shared 依赖）===
# libarchoera_subsonic.dll 由 mingw（gcc，cgo 必需）编译，动态依赖
# libgcc_s_seh-1.dll / libwinpthread-1.dll；scraper 内嵌的 mingw DLL
#（vcpkg 三角为 MSVC 构建，不含这两者）。Windows 加载器按 exe 目录解析
# 经绝对路径 open 的 subsonic DLL 的依赖，故须随包分发到 exe 根目录。
# 搜索 choco mingw 安装目录（CI）与本机 MINGW 根（如有），找不到时静默跳过
#（subsonic 模块缺失时不影响其余模块构建）。
set(MINGW_RUNTIME_SEARCH
  "$ENV{ChocolateyInstall}/lib/mingw/tools/install/mingw64/bin"
  "C:/ProgramData/chocolatey/lib/mingw/tools/install/mingw64/bin"
  "$ENV{MINGW_ROOT}/bin"
  "$ENV{MINGW64_ROOT}/bin"
)
find_file(MINGW_RT_GCC_DLL libgcc_s_seh-1.dll
  HINTS ${MINGW_RUNTIME_SEARCH} NO_DEFAULT_PATH)
find_file(MINGW_RT_PTHREAD_DLL libwinpthread-1.dll
  HINTS ${MINGW_RUNTIME_SEARCH} NO_DEFAULT_PATH)
find_file(MINGW_RT_STDCXX_DLL libstdc++-6.dll
  HINTS ${MINGW_RUNTIME_SEARCH} NO_DEFAULT_PATH)
set(MINGW_RUNTIME_DLLS "")
foreach(_d IN ITEMS MINGW_RT_GCC_DLL MINGW_RT_PTHREAD_DLL MINGW_RT_STDCXX_DLL)
  if(${_d})
    list(APPEND MINGW_RUNTIME_DLLS "${${_d}}")
  endif()
endforeach()
if(MINGW_RUNTIME_DLLS)
  install(FILES ${MINGW_RUNTIME_DLLS}
    DESTINATION "${CMAKE_INSTALL_PREFIX}"
    COMPONENT Runtime)
endif()
