// 本地曲库搜索索引器 P5 性能 harness —— 查询延迟（同一进程内循环）。
//
// 目的：在**单个进程内**对 search.db 反复执行一组代表查询，测量 P50 / P95，
// 排除 `dotnet run` / 进程启动开销（设计文档 §7.3 / §7.4）。
//
// 直接复用生产 FtsStore、Analyzer、SearchQuery / SearchSession（查询编译器同源编译），
// 不复制逻辑、不新增依赖。A/B 两组（拼音列开 / 关）各跑一次。
//
// 用法：
//   qbench --index <search.db> --label A --pinyin on|off
//          [--repeats 60] [--warmup 5] [--limit 50] [--out <json>]
// 退出码：0 成功；1 参数 / 运行错误。

using System.Diagnostics;
using System.Text.Json;
using Archoera.Indexer;
using Archoera.Text;

namespace Archoera.Indexer.Bench;

/// <summary>查询延迟测量 harness（in-process，排除进程启动开销）。</summary>
public static class QueryBench
{
    /// <summary>
    /// 代表查询（§8）：中文子串 / 中文短语 / 繁体 / **中文 + 首字母支** / 拼音连写 /
    /// **单音节拼音前缀** / 首字母 / 拉丁前缀 / 拉丁整词，以及兜底趟
    /// （拼音模糊音 / 日文读音 / 拉丁拼错 / 换位 / 中文换位）。
    /// </summary>
    private sealed record QueryCase(string Name, string Text);

    private static readonly QueryCase[] Cases =
    {
        new("cjk_substring", "\u7231\u4f60"),          // 爱你
        new("cjk_phrase", "\u6211\u597d\u7231\u4f60"),  // 我好爱你
        new("traditional", "\u611b"),                    // 愛（繁体，查询侧归一）
        new("cjk_with_initials", "\u6211\u7231\u4f60"),  // 我爱你（拼音支 + 首字母支 wan）
        new("pinyin_joined", "woaini"),                  // 连写拼音 → 相邻短语
        new("pinyin_prefix", "wo"),                      // 单音节前缀（边打边搜）
        new("pinyin_initials", "wan"),
        new("latin_prefix", "love"),
        new("latin_word", "hello"),
        // 兜底趟（越靠后越慢：要依次试完全部表达式）——T1-8 模糊层优化的基线。
        new("pinyin_fuzzy", "zongguo"),                  // 平翘舌模糊音（拼音宽松趟）
        new("romaji_reading", "kurarin"),                // 拉丁 → 日文读音趟
        new("latin_typo", "helo"),                       // 漏字母（模糊趟）
        new("latin_swap", "hlelo"),                      // 相邻换位（模糊趟）
        new("cjk_swap", "\u4f26\u5468"),                 // 伦周（中文换位容错趟）
    };

    public static int Main(string[] args)
    {
        string? index = null;
        string? outPath = null;
        var label = "A";
        var pinyinMode = "on";
        var repeats = 60;
        var warmup = 5;
        var limit = 50;

        for (var i = 0; i < args.Length; i++)
        {
            switch (args[i])
            {
                case "--index" when i + 1 < args.Length: index = args[++i]; break;
                case "--label" when i + 1 < args.Length: label = args[++i]; break;
                case "--pinyin" when i + 1 < args.Length: pinyinMode = args[++i]; break;
                case "--cap" when i + 1 < args.Length: _candidateCap = int.Parse(args[++i]); break;
                case "--out" when i + 1 < args.Length: outPath = args[++i]; break;
                case "--repeats" when i + 1 < args.Length && int.TryParse(args[++i], out var r): repeats = r; break;
                case "--warmup" when i + 1 < args.Length && int.TryParse(args[++i], out var w): warmup = w; break;
                case "--limit" when i + 1 < args.Length && int.TryParse(args[++i], out var l): limit = l; break;
                case "-h" or "--help":
                    Console.Error.WriteLine(
                        "qbench --index <search.db> --label A --pinyin on|off [--repeats N] [--warmup N] "
                        + "[--limit N] [--cap N] [--out f.json]");
                    return 0;
            }
        }

        if (index is null || !File.Exists(index))
        {
            Console.Error.WriteLine("缺少有效 --index <search.db>");
            return 1;
        }
        if (repeats <= 0) repeats = 1;

        var withPinyin = !string.Equals(pinyinMode, "off", StringComparison.OrdinalIgnoreCase);

        using var store = new FtsStore();
        store.Open(index, build: false);

        var queries = new List<object>();
        var pooled = new List<double>(Cases.Length * repeats);

        foreach (var q in Cases)
        {
            var passes = BuildPasses(q, store.LoadFuzzyVocabulary());
            var expr = passes[0];
            var rows = 0;
            var usedPasses = 1;

            for (var i = 0; i < warmup; i++)
                (rows, usedPasses) = RunProgressive(store, passes, limit);

            var samples = new List<double>(repeats);
            for (var i = 0; i < repeats; i++)
            {
                var t0 = Stopwatch.GetTimestamp();
                (rows, usedPasses) = RunProgressive(store, passes, limit);
                var t1 = Stopwatch.GetTimestamp();
                samples.Add((t1 - t0) * 1000.0 / Stopwatch.Frequency);
            }

            samples.Sort();
            pooled.AddRange(samples);

            var p50 = Percentile(samples, 0.50);
            var p95 = Percentile(samples, 0.95);
            var mean = samples.Count > 0 ? samples.Sum() / samples.Count : 0.0;

            queries.Add(new
            {
                name = q.Name,
                text = q.Text,
                match = expr,
                passes = usedPasses,
                rows,
                count = samples.Count,
                p50_ms = Round(p50),
                p95_ms = Round(p95),
                mean_ms = Round(mean),
                min_ms = Round(samples.Count > 0 ? samples[0] : 0.0),
                max_ms = Round(samples.Count > 0 ? samples[^1] : 0.0),
            });

            Console.WriteLine(
                $"[{label}] {q.Name,-16} rows={rows,6} passes={usedPasses} " +
                $"p50={p50,8:F3}ms p95={p95,8:F3}ms  expr={expr}");
        }

        pooled.Sort();
        var pooledP50 = Percentile(pooled, 0.50);
        var pooledP95 = Percentile(pooled, 0.95);
        var pooledMean = pooled.Count > 0 ? pooled.Sum() / pooled.Count : 0.0;

        var report = new
        {
            label,
            index = Path.GetFullPath(index),
            pinyin = withPinyin,
            candidateCap = _candidateCap,
            repeats,
            warmup,
            limit,
            queries,
            pooled = new
            {
                count = pooled.Count,
                p50_ms = Round(pooledP50),
                p95_ms = Round(pooledP95),
                mean_ms = Round(pooledMean),
                max_ms = Round(pooled.Count > 0 ? pooled[^1] : 0.0),
            },
        };

        var json = JsonSerializer.Serialize(report, new JsonSerializerOptions { WriteIndented = true });
        if (outPath is not null) File.WriteAllText(outPath, json);
        Console.WriteLine(json);

        return 0;
    }

    /// <summary>候选上限（**历史参数，已不再生效**；`--cap` 仍可传但结果不受影响）。</summary>
    private static int _candidateCap = FtsStore.DefaultCandidateCap;

    private static int RunOnce(FtsStore store, string matchExpr, int limit)
        => store.Search(matchExpr, limit, 0, _candidateCap).Count;

    /// <summary>
    /// 按生产短路语义依次跑各趟，返回（命中数, 实际执行趟数）：**第一个非空趟即停**
    /// （生产侧命中 ≥1 即短路，见 `SearchIndex.search`）。
    /// </summary>
    private static (int Rows, int Passes) RunProgressive(
        FtsStore store, List<string> passes, int limit)
    {
        for (var i = 0; i < passes.Count; i++)
        {
            var rows = RunOnce(store, passes[i], limit);
            if (rows > 0) return (rows, i + 1);
        }
        return (0, passes.Count);
    }

    /// <summary>
    /// 构造**渐进查询趟**——直接调用生产查询编译器，顺序与 <see cref="SearchSession.Execute"/>
    /// 完全一致（主趟 → 拼音宽松 → 日文读音 → 跨语言 → 模糊 → 中文容错）。
    /// 生产侧命中 ≥1 即短路，本 harness 用「第一个非空趟」复现同一行为与代价。
    /// </summary>
    private static List<string> BuildPasses(QueryCase q, IReadOnlyList<string> vocabulary)
    {
        var tokens = Analyzer.AnalyzeQuery(q.Text);
        var passes = new List<string>();
        void Add(string? expression)
        {
            if (expression is not null) passes.Add(expression);
        }

        Add(SearchQuery.CompileMatch(tokens, metadata: false));
        Add(SearchQuery.CompilePinyinLoose(tokens, metadata: false));
        Add(SearchQuery.CompileReading(tokens, metadata: false));
        Add(SearchQuery.CompileCrossLingual(tokens, metadata: false));
        if (SearchQuery.HasFuzzyEligible(tokens))
            Add(SearchQuery.CompileFuzzy(tokens, metadata: false, vocabulary));

        // 中文容错趟：与 SearchSession 一样，把每个变体的主趟表达式 OR 进同一趟。
        var compiled = new List<string>();
        foreach (var variant in SearchQuery.CjkVariants(q.Text))
        {
            var expression = SearchQuery.CompileMatch(Analyzer.AnalyzeQuery(variant), metadata: false);
            if (expression is not null) compiled.Add(expression);
        }
        if (compiled.Count > 0)
            passes.Add(compiled.Count == 1
                ? compiled[0]
                : "(" + string.Join(" OR ", compiled) + ")");

        if (passes.Count == 0) passes.Add("\"\"");
        return passes;
    }

    /// <summary>最近秩（nearest-rank）百分位；列表须已升序。</summary>
    private static double Percentile(List<double> sorted, double p)
    {
        if (sorted.Count == 0) return 0.0;
        var idx = (int)Math.Ceiling(p * sorted.Count) - 1;
        if (idx < 0) idx = 0;
        if (idx >= sorted.Count) idx = sorted.Count - 1;
        return sorted[idx];
    }

    private static double Round(double v) => Math.Round(v, 4, MidpointRounding.AwayFromZero);
}
