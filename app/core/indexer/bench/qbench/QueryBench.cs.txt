// 本地曲库搜索索引器 P5 性能 harness —— 查询延迟（同一进程内循环）。
//
// 目的：在**单个进程内**对 search.db 反复执行一组代表查询，测量 P50 / P95，
// 排除 `dotnet run` / 进程启动开销（设计文档 §7.3 / §7.4）。
//
// 直接复用生产 FtsStore（同一 MATCH + bm25 权重路径）与 Analyzer（查询侧同构），
// 不复制逻辑、不新增依赖。A/B 两组（拼音列开 / 关）各跑一次。
//
// 用法：
//   qbench --index <search.db> --label A --pinyin on|off
//          [--repeats 60] [--warmup 5] [--limit 50] [--out <json>]
// 退出码：0 成功；1 参数 / 运行错误。

using System.Diagnostics;
using System.Text.Json;
using Archoera.Indexer;
using Archoera.Text;

namespace Archoera.Indexer.Bench;

/// <summary>查询延迟测量 harness（in-process，排除进程启动开销）。</summary>
public static class QueryBench
{
    /// <summary>
    /// 代表查询（§8）：中文子串 / 中文短语 / 繁体 / **中文 + 首字母支** / 拼音连写 /
    /// **单音节拼音前缀** / 首字母 / 拉丁前缀 / 拉丁整词。
    /// </summary>
    private sealed record QueryCase(string Name, string Text);

    private static readonly QueryCase[] Cases =
    {
        new("cjk_substring", "\u7231\u4f60"),          // 爱你
        new("cjk_phrase", "\u6211\u597d\u7231\u4f60"),  // 我好爱你
        new("traditional", "\u611b"),                    // 愛（繁体，查询侧归一）
        new("cjk_with_initials", "\u6211\u7231\u4f60"),  // 我爱你（拼音支 + 首字母支 wan）
        new("pinyin_joined", "woaini"),                  // 连写拼音 → 相邻短语
        new("pinyin_prefix", "wo"),                      // 单音节前缀（边打边搜）
        new("pinyin_initials", "wan"),
        new("latin_prefix", "love"),
        new("latin_word", "hello"),
    };

    public static int Main(string[] args)
    {
        string? index = null;
        string? outPath = null;
        var label = "A";
        var pinyinMode = "on";
        var repeats = 60;
        var warmup = 5;
        var limit = 50;

        for (var i = 0; i < args.Length; i++)
        {
            switch (args[i])
            {
                case "--index" when i + 1 < args.Length: index = args[++i]; break;
                case "--label" when i + 1 < args.Length: label = args[++i]; break;
                case "--pinyin" when i + 1 < args.Length: pinyinMode = args[++i]; break;
                case "--cap" when i + 1 < args.Length: _candidateCap = int.Parse(args[++i]); break;
                case "--out" when i + 1 < args.Length: outPath = args[++i]; break;
                case "--repeats" when i + 1 < args.Length && int.TryParse(args[++i], out var r): repeats = r; break;
                case "--warmup" when i + 1 < args.Length && int.TryParse(args[++i], out var w): warmup = w; break;
                case "--limit" when i + 1 < args.Length && int.TryParse(args[++i], out var l): limit = l; break;
                case "-h" or "--help":
                    Console.Error.WriteLine(
                        "qbench --index <search.db> --label A --pinyin on|off [--repeats N] [--warmup N] "
                        + "[--limit N] [--cap N] [--out f.json]");
                    return 0;
            }
        }

        if (index is null || !File.Exists(index))
        {
            Console.Error.WriteLine("缺少有效 --index <search.db>");
            return 1;
        }
        if (repeats <= 0) repeats = 1;

        var withPinyin = !string.Equals(pinyinMode, "off", StringComparison.OrdinalIgnoreCase);

        using var store = new FtsStore();
        store.Open(index, build: false);

        var queries = new List<object>();
        var pooled = new List<double>(Cases.Length * repeats);

        foreach (var q in Cases)
        {
            var passes = BuildPasses(q, withPinyin);
            var expr = passes[0];
            var rows = 0;
            var usedPasses = 1;

            for (var i = 0; i < warmup; i++)
                (rows, usedPasses) = RunProgressive(store, passes, limit);

            var samples = new List<double>(repeats);
            for (var i = 0; i < repeats; i++)
            {
                var t0 = Stopwatch.GetTimestamp();
                (rows, usedPasses) = RunProgressive(store, passes, limit);
                var t1 = Stopwatch.GetTimestamp();
                samples.Add((t1 - t0) * 1000.0 / Stopwatch.Frequency);
            }

            samples.Sort();
            pooled.AddRange(samples);

            var p50 = Percentile(samples, 0.50);
            var p95 = Percentile(samples, 0.95);
            var mean = samples.Count > 0 ? samples.Sum() / samples.Count : 0.0;

            queries.Add(new
            {
                name = q.Name,
                text = q.Text,
                match = expr,
                passes = usedPasses,
                rows,
                count = samples.Count,
                p50_ms = Round(p50),
                p95_ms = Round(p95),
                mean_ms = Round(mean),
                min_ms = Round(samples.Count > 0 ? samples[0] : 0.0),
                max_ms = Round(samples.Count > 0 ? samples[^1] : 0.0),
            });

            Console.WriteLine(
                $"[{label}] {q.Name,-16} rows={rows,6} passes={usedPasses} " +
                $"p50={p50,8:F3}ms p95={p95,8:F3}ms  expr={expr}");
        }

        pooled.Sort();
        var pooledP50 = Percentile(pooled, 0.50);
        var pooledP95 = Percentile(pooled, 0.95);
        var pooledMean = pooled.Count > 0 ? pooled.Sum() / pooled.Count : 0.0;

        var report = new
        {
            label,
            index = Path.GetFullPath(index),
            pinyin = withPinyin,
            candidateCap = _candidateCap,
            repeats,
            warmup,
            limit,
            queries,
            pooled = new
            {
                count = pooled.Count,
                p50_ms = Round(pooledP50),
                p95_ms = Round(pooledP95),
                mean_ms = Round(pooledMean),
                max_ms = Round(pooled.Count > 0 ? pooled[^1] : 0.0),
            },
        };

        var json = JsonSerializer.Serialize(report, new JsonSerializerOptions { WriteIndented = true });
        if (outPath is not null) File.WriteAllText(outPath, json);
        Console.WriteLine(json);

        return 0;
    }

    /// <summary>生产候选上限（与 Dart `SearchIndex.defaultCandidateCap` 一致；`--cap 0` 关闭）。</summary>
    private static int _candidateCap = FtsStore.DefaultCandidateCap;

    private static int RunOnce(FtsStore store, string matchExpr, int limit)
        => store.Search(matchExpr, limit, 0, _candidateCap).Count;

    /// <summary>
    /// 按生产短路语义依次跑各趟，返回（命中数, 实际执行趟数）：**第一个非空趟即停**
    /// （生产侧命中 ≥1 即短路，见 `SearchIndex.search`）。
    /// </summary>
    private static (int Rows, int Passes) RunProgressive(
        FtsStore store, List<string> passes, int limit)
    {
        for (var i = 0; i < passes.Count; i++)
        {
            var rows = RunOnce(store, passes[i], limit);
            if (rows > 0) return (rows, i + 1);
        }
        return (0, passes.Count);
    }

    /// <summary>
    /// 构造**渐进查询趟**——与生产查询路径同构（`search_query.dart` 的 `compileMatch` /
    /// `compilePinyinLoose` / `compileCrossLingual` + `SearchIndex.search` 的短路语义）：
    ///   1. **主趟**：词法支（连续 CJK bigram 引号短语 + unigram `AND`，末拉丁/数字 ≥3 字符
    ///      加 `*`）+ 拼音支（单音节 ≥3 字符前缀；**≥2 音节相邻短语**）+ 首字母支
    ///      （`{initials_meta_tok initials_lyrics_tok} : "x"*`，≥2 字符前缀）——各支 `OR`；
    ///   2. **拼音宽松兜底趟**（主趟 0 命中时）：音节位置无关 `AND`（恢复漏字 / 换位召回）；
    ///   3. **日文读音兜底趟**（v5，仍 0 命中时）：拉丁词元 → 字母 bigram `AND` 到读音列；
    ///   4. **跨语言兜底趟**（仍 0 命中时）：T1 候选 `OR`。
    /// 生产侧命中 ≥1 即短路，本 harness 用「第一个非空趟」复现同一行为与代价。
    /// </summary>
    private static List<string> BuildPasses(QueryCase q, bool withPinyin)
    {
        var analyzed = Analyzer.AnalyzeQuery(q.Text);
        var passes = new List<string>();

        var branches = new List<string>();
        var lexExpr = LexicalExpr(analyzed.Lexical.Where(t => t.Length > 0).ToList());
        if (lexExpr is not null) branches.Add(lexExpr);
        if (withPinyin)
        {
            var pinyinExpr = PinyinExpr(analyzed.Pinyin);
            if (pinyinExpr is not null) branches.Add(pinyinExpr);
            var initialsExpr = InitialsExpr(analyzed.PinyinInitials);
            if (initialsExpr is not null) branches.Add(initialsExpr);
        }
        if (branches.Count > 0)
            passes.Add(branches.Count == 1 ? branches[0] : $"({string.Join(" OR ", branches)})");

        // 拼音宽松兜底（位置无关 AND；末音节 ≥3 字符前缀）
        if (withPinyin && analyzed.Pinyin.Count >= 2)
        {
            var syllables = analyzed.Pinyin.Where(t => t.Trim().Length > 0).ToList();
            var parts = new List<string>(syllables.Count);
            for (var i = 0; i < syllables.Count; i++)
                parts.Add(Quote(syllables[i]) +
                          (i == syllables.Count - 1 && syllables[i].Length >= MinPrefixTokenLength
                              ? "*" : string.Empty));
            passes.Add($"({LB}{PinyinColumns}{RB} : ({string.Join(" AND ", parts)}))");
        }

        // 日文读音兜底（v5）：拉丁 / 数字词元 → 字母 bigram AND（与生产 compileReading 同构）
        var readingParts = new List<string>();
        foreach (var token in analyzed.Lexical)
        {
            var lower = token.ToLowerInvariant();
            if (lower.Length < 2) continue;
            if (lower.EnumerateRunes().Any(r =>
                    r.Value <= char.MaxValue &&
                    (ScriptRanges.IsCjk((char)r.Value) || ScriptRanges.IsSubstringScript((char)r.Value))))
                continue;
            // 单个引号短语 = 相邻 bigram 序列（与生产 compileReading 的短语语义一致）
            var bigrams = new List<string>();
            for (var k = 0; k + 1 < lower.Length; k++) bigrams.Add(lower.Substring(k, 2));
            if (bigrams.Count > 0) readingParts.Add(Quote(string.Join(' ', bigrams)));
        }
        if (readingParts.Count > 0)
            passes.Add($"({LB}{ReadingColumns}{RB} : ({string.Join(" AND ", readingParts)}))");

        var crossExpr = CrossLingualExpr(analyzed.CrossLingual);
        if (crossExpr is not null) passes.Add(crossExpr);

        if (passes.Count == 0) passes.Add("\"\"");
        return passes;
    }

    /// <summary>前缀（边打边搜）的最小词元长度：与生产侧 <c>minPrefixTokenLength</c> 一致。</summary>
    private const int MinPrefixTokenLength = 3;

    /// <summary>首字母前缀的最小长度：与生产侧 <c>minInitialsPrefixLength</c> 一致。</summary>
    private const int MinInitialsPrefixLength = 2;

    /// <summary>词法支：连续 CJK bigram 合并为引号短语，其余逐个 `AND`；末拉丁 / 数字加前缀。</summary>
    private static string? LexicalExpr(IReadOnlyList<string> tokens)
    {
        if (tokens.Count == 0) return null;

        var chunks = new List<string>();
        var i = 0;
        while (i < tokens.Count)
        {
            if (IsCjkBigram(tokens[i]))
            {
                var j = i;
                while (j < tokens.Count && IsCjkBigram(tokens[j])) j++;
                var run = tokens.Skip(i).Take(j - i).ToList();
                chunks.Add(Quote(run.Count >= 2 ? string.Join(' ', run) : run[0]));
                i = j;
            }
            else
            {
                chunks.Add(Quote(tokens[i]));
                i++;
            }
        }

        var prefixLast = IsLatinOrDigit(tokens[^1]) && tokens[^1].Length >= MinPrefixTokenLength;
        var parts = new List<string>(chunks.Count);
        for (var k = 0; k < chunks.Count; k++)
            parts.Add(k == chunks.Count - 1 && prefixLast ? chunks[k] + "*" : chunks[k]);
        return string.Join(" AND ", parts);
    }

    /// <summary>拼音支：≥2 音节相邻短语 `"wo ai ni"`；单音节 ≥3 字符前缀 `"hao"*`（否则精确）；空返回 null。</summary>
    private static string? PinyinExpr(IReadOnlyList<string> tokens)
    {
        var clean = tokens.Where(t => t.Trim().Length > 0).ToList();
        if (clean.Count == 0) return null;
        if (clean.Count == 1)
        {
            var only = clean[0];
            var body = Quote(only) + (only.Length >= MinPrefixTokenLength ? "*" : string.Empty);
            return $"({{{PinyinColumns}}} : {body})";
        }
        return $"({{{PinyinColumns}}} : {Quote(string.Join(' ', clean))})";
    }

    /// <summary>FTS5 列过滤定界符（`{col1 col2} : expr`）。</summary>
    private const string LB = "{";
    private const string RB = "}";

    /// <summary>拼音列集合（v4：元数据段 / 歌词段分列）。</summary>
    private const string PinyinColumns = "pinyin_meta_tok pinyin_lyrics_tok";

    /// <summary>首字母列集合（v4：元数据段 / 歌词段分列）。</summary>
    private const string InitialsColumns = "initials_meta_tok initials_lyrics_tok";

    /// <summary>日文读音列集合（v5：假名 / 汉字读音 → 罗马字字母 bigram）。</summary>
    private const string ReadingColumns = "reading_meta_tok";

    /// <summary>首字母支：列过滤到首字母两列（v4 分列）；末段 ≥2 字符前缀；空 / null 返回 null。</summary>
    private static string? InitialsExpr(string? initials)
    {
        if (string.IsNullOrWhiteSpace(initials)) return null;
        var parts = initials.Split(' ', StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length == 0) return null;
        var body = string.Join(" AND ", parts.Select(
            (t, i) => Quote(t) + (i == parts.Length - 1 && t.Length >= MinInitialsPrefixLength
                ? "*" : string.Empty)));
        return $"({{{InitialsColumns}}} : {body})";
    }


    /// <summary>跨语言支（T1）：候选逐个加引号后 `OR`（候选间 OR）；空返回 null。</summary>
    private static string? CrossLingualExpr(IReadOnlyList<string> candidates)
    {
        var clean = candidates.Where(t => t.Trim().Length > 0).ToList();
        if (clean.Count == 0) return null;
        return clean.Count == 1
            ? Quote(clean[0])
            : "(" + string.Join(" OR ", clean.Select(Quote)) + ")";
    }

    private static bool IsCjkBigram(string token)
    {
        var runes = token.EnumerateRunes().ToList();
        return runes.Count == 2 && runes.All(r => r.Value <= char.MaxValue && Analyzer.IsCjk((char)r.Value));
    }

    private static string Quote(string token) => "\"" + token.Replace("\"", "\"\"") + "\"";

    private static bool IsLatinOrDigit(string token)
    {
        if (token.Length == 0) return false;
        foreach (var c in token)
        {
            if (!((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9'))) return false;
        }
        return true;
    }

    /// <summary>最近秩（nearest-rank）百分位；列表须已升序。</summary>
    private static double Percentile(List<double> sorted, double p)
    {
        if (sorted.Count == 0) return 0.0;
        var idx = (int)Math.Ceiling(p * sorted.Count) - 1;
        if (idx < 0) idx = 0;
        if (idx >= sorted.Count) idx = sorted.Count - 1;
        return sorted[idx];
    }

    private static double Round(double v) => Math.Round(v, 4, MidpointRounding.AwayFromZero);
}
