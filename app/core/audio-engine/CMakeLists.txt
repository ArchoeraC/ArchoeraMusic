# server/audio-engine/CMakeLists.txt
#
# Phase 4：FFmpeg 解码 + swresample 重采样 + EQ + loudness + limiter
#          + tempo (Rust signalsmith-stretch) + FFT + Opus 编码 + OGG 封装
#
# 构建：
#   cmake -B build -DCMAKE_BUILD_TYPE=Release
#   cmake --build build -j$(nproc)
#
# 产物：
#   build/archoera-audio-engine  (独立 CLI 二进制)
#
# 依赖（Fedora 42）：
#   dnf install ffmpeg-devel gcc cmake cargo rustc

cmake_minimum_required(VERSION 3.16)

project(archoera-audio-engine
    VERSION 0.3.0
    LANGUAGES C
)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# ── 编译选项 ────────────────────────────────────────────────
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(
        -Wall -Wextra -Wno-unused-parameter
        -O2
        -fvisibility=hidden
    )
endif()

# ── FFmpeg 依赖 ──────────────────────────────────────────────
# 本文件仅服务 Linux/macOS（系统包 + pkg-config）。
# Windows 构建已彻底独立：使用 app/core/build_windows.bat（vcpkg + cl.exe 直编），
# 不经过 CMake，因此这里不再维护任何 WIN32/MSVC 分支。
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED
    libavformat
    libavcodec
    libavutil
    libswresample
)

# ── FFmpeg 运行库内嵌（Linux 自包含，防系统 FFmpeg major bump 破坏）─
# FFmpeg 每次 major 升级都 bump soname（如 libavformat.so.62→.63），系统上
# 所有链接旧 soname 的预编译二进制随即无法加载（官方声明要求下游全部重编）。
# 为避免「系统升级 → 应用坏掉」，构建时把链接到的 FFmpeg 运行库实体拷入
# build/ffmpeg/，由 app/linux 一并装进 bundle/native/；引擎 RUNPATH=$ORIGIN
# 使加载器优先解析自带库，与系统 FFmpeg 完全解耦。
# 注意：macOS 的 dylib install_name 是绝对路径，内嵌需 install_name_tool
# 后处理，本块仅服务 Linux（ELF 按 soname 纯文件名解析）。
if(NOT APPLE)
  file(GLOB FFMPEG_RUNTIME_LIBS
    "${FFMPEG_LIBRARY_DIRS}/libavformat.so.*"
    "${FFMPEG_LIBRARY_DIRS}/libavcodec.so.*"
    "${FFMPEG_LIBRARY_DIRS}/libavutil.so.*"
    "${FFMPEG_LIBRARY_DIRS}/libswresample.so.*")
  if(FFMPEG_RUNTIME_LIBS)
    file(COPY ${FFMPEG_RUNTIME_LIBS} DESTINATION ${CMAKE_CURRENT_BINARY_DIR}/ffmpeg)
    message(STATUS "FFmpeg 内嵌运行库 → ${CMAKE_CURRENT_BINARY_DIR}/ffmpeg/")
  else()
    message(WARNING "FFmpeg 运行库未找到（${FFMPEG_LIBRARY_DIRS}），bundle 将依赖系统 FFmpeg")
  endif()
endif()

# ── 自研 Zig 解码内核（EraAudio 原生优先解码；docs/audio-kernel-zig.md §16）─
# libarchoera_kernel.a（zig build 静态库）经 include/kernel_bridge.h 的 zk_* C ABI
# 被 src/native_decoder.c 调用，作为 FFmpeg 的渐进替换后端：
#   - 源树已有预构建 zig-out/lib/libarchoera_kernel.a → 直接复用（拷入 build/，
#     每次构建 copy_if_different 感知更新，避免重跑整库 zig 编译）；
#   - 无预构建产物但存在 zig → add_custom_command 以 --prefix 现构建；
#   - 两者皆无 → HAS_ARCHOERA_KERNEL 关闭，EraAudio 模式回退 FFmpeg（功能不受损）。
# 内核与 C 壳同用 c_allocator（宿主 CRT malloc/free），静态库无跨 CRT 问题。
set(ZK_SRC_LIB ${CMAKE_CURRENT_SOURCE_DIR}/zig-out/lib/libarchoera_kernel.a)
set(ZK_LIB ${CMAKE_CURRENT_BINARY_DIR}/libarchoera_kernel.a)
set(ZK_OUT_DIR ${CMAKE_CURRENT_BINARY_DIR}/archoera_kernel-out)
set(HAS_ARCHOERA_KERNEL FALSE)

if(EXISTS ${ZK_SRC_LIB})
    add_custom_command(
        OUTPUT ${ZK_LIB}
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ZK_SRC_LIB} ${ZK_LIB}
        DEPENDS ${ZK_SRC_LIB}
        COMMENT "复用预构建 Zig 解码内核（archoera_kernel）"
        VERBATIM
    )
    add_custom_target(archoera_kernel_lib DEPENDS ${ZK_LIB})
    set(HAS_ARCHOERA_KERNEL TRUE)
else()
    find_program(ZIG_EXE zig)
    if(ZIG_EXE)
        # 现构建：--prefix 指向本 build 目录，避免写源树 zig-out/
        add_custom_command(
            OUTPUT ${ZK_LIB}
            COMMAND ${ZIG_EXE} build --prefix ${ZK_OUT_DIR} -Doptimize=ReleaseFast
            COMMAND ${CMAKE_COMMAND} -E copy
                ${ZK_OUT_DIR}/lib/libarchoera_kernel.a ${ZK_LIB}
            WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
            COMMENT "构建 Zig 解码内核（archoera_kernel）"
            VERBATIM
        )
        add_custom_target(archoera_kernel_lib DEPENDS ${ZK_LIB})
        set(HAS_ARCHOERA_KERNEL TRUE)
        message(STATUS "Zig 解码内核由 CMake 现构建（zig build）")
    else()
        message(WARNING
            "zig 不可用且无预构建 libarchoera_kernel.a — EraAudio 模式将回退 FFmpeg")
    endif()
endif()

# ── Rust tempo 库 ──────────────────────────────────────────
set(TEMPO_RS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/tempo-rs)
# Rust staticlib 命名（非 Windows 平台为 .a；MSVC 的 .lib 仅 Windows
# 一站式脚本 app/core/build_windows.bat 处理，本文件不涉及）
set(TEMPO_LIB ${CMAKE_CURRENT_BINARY_DIR}/libaudio_tempo.a)
set(TEMPO_RUST_SRC libaudio_tempo.a)

# 尝试查找 cargo
find_program(CARGO_EXE cargo)
if(CARGO_EXE)
    # 构建 Rust staticlib
    add_custom_command(
        OUTPUT ${TEMPO_LIB}
        COMMAND ${CARGO_EXE} build --release --target-dir ${CMAKE_CURRENT_BINARY_DIR}/tempo-target
        COMMAND ${CMAKE_COMMAND} -E copy
            ${CMAKE_CURRENT_BINARY_DIR}/tempo-target/release/${TEMPO_RUST_SRC}
            ${TEMPO_LIB}
        WORKING_DIRECTORY ${TEMPO_RS_DIR}
        COMMENT "Building Rust tempo library (audio_tempo)"
        VERBATIM
    )
    add_custom_target(tempo_lib DEPENDS ${TEMPO_LIB})
    set(TEMPO_AVAILABLE TRUE)
else()
    message(WARNING "cargo not found — tempo (变速变调) 功能将不可用，请在构建环境安装 Rust")
    set(TEMPO_AVAILABLE FALSE)
endif()

# 输出设备枚举平台后端（编译期单选；与 audio_output_platform.h 的约定一致）
if(WIN32)
    list(APPEND AUDIO_ENGINE_SOURCES src/audio_output_windows.c)
elseif(APPLE)
    list(APPEND AUDIO_ENGINE_SOURCES src/audio_output_macos.c)
elseif(UNIX)
    list(APPEND AUDIO_ENGINE_SOURCES src/audio_output_linux.c)
else()
    list(APPEND AUDIO_ENGINE_SOURCES src/audio_output_stub.c)

set(AUDIO_ENGINE_EXTRA_SOURCES
    src/tempo.c
)

set(AUDIO_ENGINE_INCLUDES
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# ── 数学库 ──────────────────────────────────────────────────
find_library(MATH_LIBRARY m)

# ── 静态库（供测试和其他程序复用）────────────────────────────
add_library(audio_engine_static STATIC ${AUDIO_ENGINE_SOURCES})
set_target_properties(audio_engine_static PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(audio_engine_static PUBLIC
    ${AUDIO_ENGINE_INCLUDES}
    ${FFMPEG_INCLUDE_DIRS}
)
target_link_libraries(audio_engine_static PUBLIC ${FFMPEG_LIBRARIES})
target_link_directories(audio_engine_static PUBLIC ${FFMPEG_LIBRARY_DIRS})
target_compile_options(audio_engine_static PUBLIC ${FFMPEG_CFLAGS_OTHER})
# dlopen（libpulse 运行期加载）在旧 glibc 需 -ldl；新 glibc 已并入 libc（CMAKE_DL_LIBS 为空）
if(CMAKE_DL_LIBS)
    target_link_libraries(audio_engine_static PUBLIC ${CMAKE_DL_LIBS})
endif()
# macOS：CoreAudio/CoreFoundation（audio_output_macos.c 原生设备枚举）
if(APPLE)
    target_link_libraries(audio_engine_static PUBLIC
        "-framework CoreAudio" "-framework CoreFoundation")
endif()

# ── Zig 解码内核接入：EraAudio 原生优先路径（native_decoder.c → zk_*）────
if(HAS_ARCHOERA_KERNEL)
    target_compile_definitions(audio_engine_static PUBLIC HAS_ARCHOERA_KERNEL=1)
    # 内核静态库以 PUBLIC 透传到各可执行/共享库最终链接（静态库不合并依赖，
    # 需在最终链接阶段解析 zk_* 符号；archoera_kernel_lib 先行构建保证产物存在）
    add_dependencies(audio_engine_static archoera_kernel_lib)
    target_link_libraries(audio_engine_static PUBLIC ${ZK_LIB})
endif()

# ── 数学库（必须排在内核静态库之后）──────────────────────────────
# 内核库引用 cos/exp2/log10f/sincos/... 等 libm 符号；Ubuntu 的 gcc 默认
# 传 --as-needed，若 -lm 出现在内核库之前，扫描时尚未出现未定义符号 → libm
# 被丢弃，随后内核库才引入这些符号 → undefined reference（Arch 默认无
# --as-needed，故本机不复现）。置于内核库之后可确保符号被解析。
if(MATH_LIBRARY)
    target_link_libraries(audio_engine_static PUBLIC ${MATH_LIBRARY})
endif()

# ── FFT 共享库（Dart FFI 客户端频谱分析；EnginePaths.libfftPath）──────────
# 独立编译 src/fft.c（无外部依赖，仅 libm）。注意全局 -fvisibility=hidden
# 会隐藏所有导出符号，此目标需恢复 default 可见性，否则 Dart 侧
# DynamicLibrary.lookup('fft_create') 等将找不到符号。
add_library(fft SHARED src/fft.c src/era_log.c)
set_target_properties(fft PROPERTIES C_VISIBILITY_PRESET default)
# 全局 add_compile_options(-fvisibility=hidden) 位于编译参数末尾，会覆盖
# C_VISIBILITY_PRESET；需在此目标显式追加 default（GCC/Clang 后者生效）。
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(fft PRIVATE -fvisibility=default)
endif()
target_include_directories(fft PRIVATE ${AUDIO_ENGINE_INCLUDES})
if(MATH_LIBRARY)
    target_link_libraries(fft PRIVATE ${MATH_LIBRARY})
endif()

# ── CLI 二进制（仅 Linux/macOS：main.c 深度依赖 POSIX——
#    unistd/fcntl/getopt_long/clock_gettime 等；Windows 桌面端 FFI 库
#    archoera_mediaengine.dll 由 build_windows.bat 独立构建）──────
add_executable(archoera-audio-engine
    src/main.c
    ${AUDIO_ENGINE_EXTRA_SOURCES}
)
target_link_libraries(archoera-audio-engine PRIVATE audio_engine_static)

# 如果 Rust tempo 可用，链接
if(TEMPO_AVAILABLE)
    add_dependencies(archoera-audio-engine tempo_lib)
    # C++ 运行时按平台：AppleClang 用 libc++ / GCC 用 stdc++
    if(APPLE)
        target_link_libraries(archoera-audio-engine PRIVATE ${TEMPO_LIB} c++)
    else()
        target_link_libraries(archoera-audio-engine PRIVATE ${TEMPO_LIB} stdc++)
    endif()
    target_compile_definitions(archoera-audio-engine PRIVATE HAS_TEMPO=1)
endif()

# ── RUNPATH=$ORIGIN：bundle/native/ 平铺布局下引擎优先解析同目录内嵌
#    FFmpeg 库（独立构建链产物被 app/linux 以 install(FILES) 直接拷贝，
#    不经 CMake 安装后处理，故需链接期显式写入）。
#    注意：必须写 \$ORIGIN（CMake 转义 $）。若写成 '$ORIGIN'（单引号），
#    CMake 会把单引号原样交给链接器，RUNPATH 变成字面量 '$ORIGIN'（带引号），
#    加载器无法展开该路径 → 引擎退回系统 FFmpeg；bundle-linux-runtime.sh
#    随即将系统 FFmpeg 的整套视频/图像依赖闭包（libx264/x265/vpx/aom/jxl/
#    rsvg/icu…）内嵌进 native/，deb 体积暴涨数百 MB。
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # -z,origin：置 ELF 的 DF_ORIGIN 标志。缺该标志时加载器会忽略 $ORIGIN
  # RUNPATH（安全策略），引擎就找不到同目录内嵌的 FFmpeg 库——系统 FFmpeg
  # 版本与内嵌不一致时直接加载失败（如发行版系统 FFmpeg 升级后 soname 不符）。
  target_link_options(archoera-audio-engine PRIVATE "-Wl,-z,origin" "-Wl,-rpath,\$ORIGIN")
  target_link_options(archoera_mediaengine PRIVATE "-Wl,-z,origin" "-Wl,-rpath,\$ORIGIN")
  # -Bsymbolic-functions：mediaengine 内嵌 Zig 内核（zk_* 符号）与同目录动态
  # libarchoera_kernel.so（scanner-ffi 加载）**同名**；若后者被全局加载，mediaengine
  # 内部对 zk_* 的引用可能被抢占到另一份内核（全局态不同）→ Host/Stream 状态错位
  # 崩溃（实测 rt 变为无效小值）。绑定本地引用，杜绝符号抢占。
  target_link_options(archoera_mediaengine PRIVATE "-Wl,-Bsymbolic")
endif()
