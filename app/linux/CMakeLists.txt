# Compilation settings that should be applied to most targets.
#
# Be cautious about adding new options here, as plugins use this function by
# default. In most cases, you should add new options to specific targets instead
# of modifying this function.
function(APPLY_STANDARD_SETTINGS TARGET)
  target_compile_features(${TARGET} PUBLIC cxx_std_14)
  # -Wno-deprecated-declarations：平台桥接经 dlopen 调用的部分 GTK/GDK API
  # （窗口状态轻轮询）在新版库中被标记 G_DEPRECATED，-Werror 会把该弃用告警升级为
  # 错误，这里显式关闭该告警类（不影响其它告警）。
  target_compile_options(${TARGET} PRIVATE -Wall -Werror -Wno-deprecated-declarations)
  target_compile_options(${TARGET} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-O3>")
  target_compile_definitions(${TARGET} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:NDEBUG>")
endfunction()

# === ArchoeraMusic downloader：内嵌 cargo 编译链（一键构建）===
# flutter build / flutter run 时自动调用 cargo 产出 libarchoera_downloader.{so,dylib,dll}
# 并镜像进 bundle，无需手动先 cargo build。cargo 自身增量编译，无变更时仅 ~0.1s。
# 构建模式与 Flutter 对齐：Debug → target/debug（cargo build），
# Release/Profile → target/release（cargo build --release）。
find_program(CARGO cargo REQUIRED)
set(DOWNLOADER_CRATE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/downloader")
if(CMAKE_BUILD_TYPE MATCHES "Debug")
  set(DOWNLOADER_TARGET_DIR "${DOWNLOADER_CRATE_DIR}/target/debug")
  set(DOWNLOADER_MODE_DIR "debug")
  set(DOWNLOADER_CARGO_FLAGS "")
else()
  set(DOWNLOADER_TARGET_DIR "${DOWNLOADER_CRATE_DIR}/target/release")
  set(DOWNLOADER_MODE_DIR "release")
  set(DOWNLOADER_CARGO_FLAGS "--release")
endif()
if(WIN32)
  set(DOWNLOADER_LIB_NAME "archoera_downloader.dll")
elseif(APPLE)
  set(DOWNLOADER_LIB_NAME "libarchoera_downloader.dylib")
else()
  set(DOWNLOADER_LIB_NAME "libarchoera_downloader.so")
endif()
# 每次都执行 cargo（其自身增量），保证 Rust 源码变更后 flutter build 自动重编。
add_custom_target(downloader_cargo ALL
  COMMAND ${CARGO} build ${DOWNLOADER_CARGO_FLAGS}
  WORKING_DIRECTORY "${DOWNLOADER_CRATE_DIR}"
  COMMENT "cargo build archoera-downloader"
  VERBATIM
)
# 二进制 target 构建/安装前先完成 Rust 编译。
add_dependencies(${BINARY_NAME} downloader_cargo)

# === ArchoeraMusic archoerashell：内嵌 cargo staticlib（一键构建） ===
# 以 staticlib 形式链入 runner：main() 检测到 `archoerashell` 子命令时直接调用
# Rust 入口，命令行调用完全不初始化 Flutter/Dart（见 app/core/shell）。
#
# 该 .a 是**可执行文件的链接输入**，必须作为 `add_custom_command(OUTPUT ...)` 的
# 真实产物声明：只写 `add_custom_target` 时 Ninja 在规划阶段找不到产出规则，会报
# “missing and no known rule to make it”（本地因 target/ 已有旧产物而不炸，
# CI 干净检出 / Nix 沙箱必炸）。
set(SHELL_CRATE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/shell")
if(CMAKE_BUILD_TYPE MATCHES "Debug")
  set(SHELL_TARGET_DIR "${SHELL_CRATE_DIR}/target/debug")
  set(SHELL_CARGO_FLAGS "")
else()
  set(SHELL_TARGET_DIR "${SHELL_CRATE_DIR}/target/release")
  set(SHELL_CARGO_FLAGS "--release")
endif()
set(SHELL_STATIC_LIB "${SHELL_TARGET_DIR}/libarchoera_shell.a")
# Rust 源码 / 清单 / build.rs / ARB 文案（build.rs 从中生成 help_strings.rs）/ 内置素材
# （`assets/` 经 include_str! 编入，任一变化都要重编；Cargo 自身增量，无变更时仅 ~0.1s）。
file(GLOB_RECURSE SHELL_RUST_SOURCES CONFIGURE_DEPENDS "${SHELL_CRATE_DIR}/src/*.rs")
file(GLOB SHELL_ARB_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../lib/l10n/*.arb")
file(GLOB_RECURSE SHELL_ASSET_SOURCES CONFIGURE_DEPENDS "${SHELL_CRATE_DIR}/assets/*")
add_custom_command(
  OUTPUT "${SHELL_STATIC_LIB}"
  COMMAND ${CARGO} build ${SHELL_CARGO_FLAGS}
  WORKING_DIRECTORY "${SHELL_CRATE_DIR}"
  DEPENDS
    "${SHELL_CRATE_DIR}/Cargo.toml"
    "${SHELL_CRATE_DIR}/Cargo.lock"
    "${SHELL_CRATE_DIR}/build.rs"
    ${SHELL_RUST_SOURCES}
    ${SHELL_ARB_SOURCES}
    ${SHELL_ASSET_SOURCES}
  COMMENT "cargo build archoera-shell"
  VERBATIM
)
add_custom_target(shell_cargo ALL DEPENDS "${SHELL_STATIC_LIB}")
add_dependencies(${BINARY_NAME} shell_cargo)
target_link_libraries(${BINARY_NAME} PRIVATE "${SHELL_STATIC_LIB}")
# Rust std 在 Linux 上的系统依赖。
target_link_libraries(${BINARY_NAME} PRIVATE pthread dl m rt util)

# === ArchoeraMusic audio-engine：内嵌 CMake 构建链（一键构建）===
# flutter build / flutter run 时自动调用 cmake 产出 libfft.so /
# libarchoera_mediaengine.so / archoera-audio-engine 到 audio-engine/build/
#（与下方 OPTIONAL install 路径一致），无需手动 cmake 预构建。
# cmake 自身增量编译，无变更时仅 ~0.1s。依赖系统 ffmpeg-devel
#（pkg-config libavformat/libavcodec/libavutil/libswresample）与 cargo

  COMMAND ${AUDIO_ENGINE_CMAKE} --build "${AUDIO_ENGINE_BUILD_DIR}" -j
  COMMENT "cmake build archoera-audio-engine (libfft + mediaengine)"
  VERBATIM
)
add_dependencies(${BINARY_NAME} audio_engine_cmake)

# === ArchoeraMusic 原生产物打包（FFI 直连，降低运行时依赖）===
# 引擎模块位于 app/core/{audio-engine,scanner,scraper,subsonic}（app 内部，路径关系固定，
# 可随 app 独立部署），产物统一安装进 bundle/native/ 平铺（对齐 ffi-libs-layout 计划；
# NativeLibPaths 祖先链优先查 native/，随后回退旧 <module>/build 布局）：
#   bundle/native/{archoera-audio-engine,libarchoera_mediaengine.so,libfft.so}
#   bundle/native/{scanner-ffi.so,libe_sqlite3.so}
#   bundle/native/{libarchoera_scraper.so}
#   bundle/native/{libarchoera_subsonic.so,libarchoera_transcoder.so}
#   bundle/native/{archoera-vault}
#   bundle/native/libarchoera_downloader.so
# 产物由独立构建链产出（audio-engine/scraper: cmake; scanner: dotnet publish;
# subsonic: build.sh 即 cargo + go; downloader: cargo; vault(Linux): cargo vault-rs）。
# OPTIONAL：源产物缺失时跳过（如未构建 subsonic 仍可跑 Flutter 构建）。
set(ENGINE_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/audio-engine/build")
set(SCANNER_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/scanner/build")
set(SCRAPER_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/scraper/build")
set(SUBSONIC_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../core/subsonic/build")

install(PROGRAMS "${ENGINE_SRC_DIR}/archoera-audio-engine"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${ENGINE_SRC_DIR}/libarchoera_mediaengine.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${ENGINE_SRC_DIR}/libfft.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
# FFmpeg 内嵌运行库（audio-engine 构建时自系统 FFmpeg 拷入其 build/ffmpeg/；
# 与引擎同装 native/，引擎 RUNPATH=$ORIGIN 优先解析，系统 FFmpeg 升级不影响）。
install(DIRECTORY "${ENGINE_SRC_DIR}/ffmpeg/"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${SCANNER_SRC_DIR}/scanner-ffi.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
# 自研内核元数据桥动态库：scanner-ffi 经 $ORIGIN 加载（zk_metadata_* 无 JSON）；
# 缺失则 scanner 回退 TagLibSharp（功能正确但非内核直桥）。
install(FILES "${SCANNER_SRC_DIR}/libarchoera_kernel.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${SCANNER_SRC_DIR}/libe_sqlite3.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${SCRAPER_SRC_DIR}/libarchoera_scraper.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${SUBSONIC_SRC_DIR}/libarchoera_subsonic.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)
install(FILES "${SUBSONIC_SRC_DIR}/libarchoera_transcoder.so"
  DESTINATION "${CMAKE_INSTALL_PREFIX}/native" COMPONENT Runtime OPTIONAL)

# vault（凭据保险库）：Linux 用 vault-rs 的 **Rust 静态单文件**（+crt-static，CI 走
